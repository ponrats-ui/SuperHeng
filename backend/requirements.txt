# SuperHeng backend — Day 2 dependencies
#
# Install into the backend virtualenv:
#     cd backend
#     python -m venv .venv
#     .venv/Scripts/python -m pip install -r requirements.txt   # Windows
#     .venv/bin/python     -m pip install -r requirements.txt   # macOS / Linux
#
# Interpreter: Python 3.14.4 (verified on this machine). Every package below
# installs from a wheel with no build step — see "Verified stack" at the bottom.

# --- API service (main.py) -------------------------------------------------
fastapi>=0.143
# [standard] adds the watchfiles reloader (needed for --reload on Windows) plus
# httptools/websockets. Plain `uvicorn` has no reloader and silently ignores
# --reload, which is a confusing failure mode during development.
uvicorn[standard]>=0.54

# --- Database (Supabase / PostgreSQL) --------------------------------------
# Pulls postgrest-py, storage3, realtime and httpx.
supabase>=2.32

# --- Configuration ---------------------------------------------------------
python-dotenv>=1.2

# --- Ingestion (scraper.py / data_ingestion.py) ----------------------------
requests>=2.32
beautifulsoup4>=4.15      # HTML archive fallback parser
pandas>=3.0               # optional: CSV export in data_ingestion.py

# --- Advanced models (Day 5) -----------------------------------------------
# NOTE: statsmodels is imported lazily, inside /api/predict/advanced, not at
# module scope. `import statsmodels.tsa.arima.model` costs a measured 1738 ms on
# this machine, and a top-level import would add that to every uvicorn start,
# every --reload restart and every test run that merely imports main.
statsmodels>=0.15
# Declared as requested, and NOT yet imported by any backend module — there is no
# sklearn call in the Day 5 code. It is the baseline for the ML work the roadmap
# still has ahead: chronological train/validation splits and permutation
# importance for the LSTM tier (VISION §2.4), and rank aggregation for the
# scorecard. It pulls scipy, which statsmodels also needs, so the install cost is
# shared rather than additive. Stated plainly rather than given a token use to
# justify its presence.
scikit-learn>=1.9

# --- Development / tests (not needed to run the service) -------------------
# The Day 3 unit tests for the statistical models:
#     .venv/Scripts/python -m pytest -q
pytest>=9.1
